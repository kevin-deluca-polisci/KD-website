Drawn inline rather than through charts.py because it needs no layout decision the payload does not already fix: the x axis is a date range and the y axis is a margin. Colours come from the same --c- custom properties the rest of the site uses, so a polling line is the same green here as on the front page. The zero line is drawn and labelled because on a race page it is the only value meaningful on its own: above it, the Democrat leads. */}} {{- $series := .series -}} {{- $labels := .labels -}} {{- $order := .order -}} {{/* Extent over every series, always including zero so the baseline is on the chart even for a race nobody puts near even. */}} {{- $lo := 0.0 -}}{{- $hi := 0.0 -}}{{- $dates := slice -}} {{- range $k, $v := $series -}} {{- range $v -}} {{- $y := index . 1 -}} {{- if lt $y $lo }}{{ $lo = $y }}{{ end -}} {{- if gt $y $hi }}{{ $hi = $y }}{{ end -}} {{- $dates = $dates | append (index . 0) -}} {{- end -}} {{- end -}} {{- $pad := math.Max 1.0 (mul (sub $hi $lo) 0.08) -}} {{- $lo = sub $lo $pad -}}{{- $hi = add $hi $pad -}} {{- $span := sub $hi $lo -}} {{- $dates = sort (uniq $dates) -}} {{- $n := len $dates -}} {{- $W := 900 -}}{{- $H := 260 -}}{{- $L := 46 -}}{{- $R := 12 -}}{{- $T := 12 -}}{{- $B := 26 -}} {{- $iw := sub (sub $W $L) $R -}} {{- $ih := sub (sub $H $T) $B -}} {{- $y0 := add $T (mul (div (sub $hi 0.0) $span) $ih) -}} {{- if gt $n 1 -}}
